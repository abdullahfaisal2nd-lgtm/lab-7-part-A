#include "doubly_linked_list.h"

void DoublyLinkedList::InsertRight(BookNode* node)
{
    if (node == nullptr)
        return;

    node->left = tail;
    node->right = nullptr;

    if (tail != nullptr)
        tail->right = node;
    else
        head = node;

    tail = node;

    if (current == nullptr)
        current = node;

    count++;
}
